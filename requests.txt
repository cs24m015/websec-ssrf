## VALID

# Define the URL of the Flask app
$url = "http://localhost:5000/fetch"

# Define the JSON body for an external URL
$body = @{
    url = "http://confidential_service:5001/confidential"
} | ConvertTo-Json

# Send the POST request
$response = Invoke-RestMethod -Uri $url -Method Post -Body $body -ContentType "application/json"

# Output the response
$response.data

## Unix
curl -X POST http://localhost:5000/fetch -H "Content-Type: application/json" -d '{"url": "http://localhost:5001/confidential"}'


## INVALID

# Define the URL of the Flask app
$url = "http://localhost:5000/fetch"

# Define an empty JSON body
$body = @{} | ConvertTo-Json

# Send the POST request
$response = Invoke-RestMethod -Uri $url -Method Post -Body $body -ContentType "application/json"

# Output the response
$response


## Unix
curl -X POST http://localhost:5000/fetch -H "Content-Type: application/json" -d '{}'
